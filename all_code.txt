#include <iostream>
#include <vector>
#include <queue>
#include <climits>
#include <functional>
using namespace std;

vector<int> dijkstra(int src, vector<vector<pair<int, int>>>& graph) {
    int n = graph.size();
    vector<int> dist(n, INT_MAX);

    priority_queue<pair<int, int>,
                   vector<pair<int, int>>,
                   greater<pair<int, int>>> pq;

    dist[src] = 0;
    pq.push({0, src});

    while (!pq.empty()) {
        int d = pq.top().first;
        int u = pq.top().second;
        pq.pop();

        if (d > dist[u])
            continue;

        for (auto edge : graph[u]) {
            int v = edge.first;
            int w = edge.second;

            if (dist[u] != INT_MAX && dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }

    return dist;
}

int main() {
    int n = 5;
    vector<vector<pair<int, int>>> graph(n);

    graph[0].push_back({1, 4});
    graph[0].push_back({2, 1});
    graph[2].push_back({1, 2});
    graph[1].push_back({3, 1});
    graph[2].push_back({3, 5});
    graph[3].push_back({4, 3});

    vector<int> dist = dijkstra(0, graph);

    cout << "Shortest distances from location 0:\n";

    for (int i = 0; i < n; i++) {
        cout << "Location " << i << ": ";

        if (dist[i] == INT_MAX)
            cout << "Unreachable";
        else
            cout << dist[i];

        cout << endl;
    }

    return 0;
}

using namespace std;

vector<int> getPath(vector<int> &par, int src, int des) {
    vector<int> path;
    int cur = des;

    while (cur != -1) {
        path.push_back(cur);

        if (cur == src)
            break;

        cur = par[cur];
    }

    if (path.back() != src) {
        path.clear();
        return path;
    }

    reverse(path.begin(), path.end());
    return path;
}

#include "priorityqueue.h"
#include <algorithm>
#include <stdexcept>

void PriorityQueue::push(int node, double cost)
{
    q.push_back({node, cost});
    int i = static_cast<int>(q.size()) - 1;

    while (i > 0) {
        int parent = (i - 1) / 2;
        if (q[parent].cost <= q[i].cost)
            break;
        std::swap(q[parent], q[i]);
        i = parent;
    }
}

PQNode PriorityQueue::pop()
{
    if (q.empty())
        throw std::out_of_range("PriorityQueue::pop called on empty queue");

    PQNode result = q.front();
    q.front() = q.back();
    q.pop_back();

    int i = 0;
    const int n = static_cast<int>(q.size());
    while (true) {
        int left = 2 * i + 1;
        int right = 2 * i + 2;
        int smallest = i;

        if (left < n && q[left].cost < q[smallest].cost)
            smallest = left;
        if (right < n && q[right].cost < q[smallest].cost)
            smallest = right;
        if (smallest == i)
            break;

        std::swap(q[i], q[smallest]);
        i = smallest;
    }
    return result;
}

bool PriorityQueue::empty()
{
    return q.empty();
}

# Adaptive Emergency Evacuation Route Optimization

An intelligent emergency response navigation system built in C++ that dynamically calculates safe evacuation routes during disasters (floods, fires, road blockages) using graph theory and shortest-path algorithms.

---

## ğŸ“Œ Project Overview

During severe emergencies, standard GPS routes can become hazardous or impassable due to sudden road damage, active hazard zones, or traffic jams. Traditional static routing algorithm systems fail to adapt in real time. 

This project models a city road network as a **dynamic weighted graph**. It computes optimal routes to safety centers using **Dijkstra's Algorithm combined with a Priority Queue**. When road conditions change, the edge weights update dynamically to recalculate safe alternate paths, bypassing affected zones.

 SYSTEM ARCHITECTURE EXECUTION PIPELINE

### [1] USER INPUT

- Receives Origin Node  (Start Location)
- Receives Destination Node (Evacuation Center / Safe Zone)

â†“

### [2] ROAD NETWORK DATA

- Map converted to Graph Nodes (Location objects)
- Edges assigned dynamic weights (Road objects with travel time)

â†“

### [3] DIJKSTRA'S ALGORITHM + PRIORITY QUEUE

- Evaluates shortest path using Min-Heap
- Calculates optimal evacuation route based on current weights

â†“

### [4] INITIAL BEST ROUTE

- Primary evacuation route established and displayed to user

â†“

### [5] ROAD CONDITION UPDATE (DYNAMIC TRIGGER)

- Incident detected (e.g., road blockage, flooding, heavy risk)
- Affected Road status changed to Blocked / Hazard level scaled

â†“

### [6] ALTERNATE ROUTE RE-CALCULATION

- Dynamic route optimization triggered
- New safe path generated bypassing affected edges

---

## âš™ï¸ Key Features

* **Dynamic Edge Weighting:** Calculates dynamic cost based on base travel time, risk/hazard multipliers, and total road closure flags[cite: 1].
* **Safety Center Routing:** Prioritizes routing affected citizens toward designated evacuation zones and medical centers[cite: 1].
* **Real-time Re-routing:** Reroutes traffic away from blocked or risky edges dynamically without rebuilding the entire graph[cite: 1].
* **Object-Oriented Design:** Built using clean, modular OOP architecture in C++[cite: 1].

---

## ğŸ› ï¸ Data Structures & Algorithms Used

* **Graph Representation:** Adjacency List for space-efficient storage of nodes and edges[cite: 1].
* **Priority Queue / Min-Heap:** Fast node extraction with $O(\log V)$ efficiency during shortest path lookup[cite: 1].
* **Dijkstraâ€™s Algorithm:** Evaluates total path costs dynamically to compute safe evacuation routes[cite: 1].
* **Object-Oriented Programming (OOP):** Core entities structured via encapsulation (`Location`, `Road`, `Graph`, `RouteOptimizer`)[cite: 1].

#include "priorityqueue.h"
void PriorityQueue::push(int node,int cost)
{
PQNode x;
x.node=node;
x.cost=cost;

q.push_back(x);

int i=q.size()-1;

while(i>0)
{
int p=(i-1)/2;

if(q[p].cost<=q[i].cost)
break;

swap(q[p],q[i]);
i=p;
}
}

PQNode PriorityQueue::pop()
{
PQNode x=q[0];

q[0]=q.back();
q.pop_back();

int i=0;

while(true)
{
int l=2*i+1;
int r=2*i+2;
int s=i;
if(l<q.size()&&q[l].cost<q[s].cost)
s=l;
if(r<q.size()&&q[r].cost<q[s].cost)
s=r;
if(s==i)
break;
swap(q[i],q[s]);
i=s;
}
return x;
}
bool PriorityQueue::empty()
{
return q.empty();
}

vector<int> getPath(vector<int> &par,int src,int des)
{
vector<int> path;
int cur=des;
while(cur!=-1)
{
path.push_back(cur);
if(cur==src)
break;
cur=par[cur];
}
if(path.back()!=src)
{
path.clear();
return path;
}
reverse(path.begin(),path.end());
return path;
}

#include "Graph.h"
#include <iostream>
#include <fstream>
#include <string>
#include <regex>
#include <stdexcept>

using namespace std;

Graph loadGraphFromJSON(const string& filename)
{
    ifstream file(filename);

    if (!file.is_open())
    {
        throw runtime_error("Cannot open JSON file: " + filename);
    }

    string content;
    string line;

    while (getline(file, line))
    {
        content += line + "\n";
    }

    // Read total number of locations
    regex locationsPattern(
        R"("totalLocations"\s*:\s*(\d+))"
    );

    smatch match;

    if (!regex_search(content, match, locationsPattern))
    {
        throw runtime_error("totalLocations not found");
    }

    int totalLocations = stoi(match[1].str());

    if (totalLocations <= 0)
    {
        throw runtime_error("Invalid number of locations");
    }

    Graph graph(totalLocations);

    // Find the roads array
    size_t roadsStart = content.find("\"roads\"");

    if (roadsStart == string::npos)
    {
        throw runtime_error("Roads array not found");
    }

    size_t arrayStart = content.find('[', roadsStart);
    size_t arrayEnd = content.find(']', arrayStart);

    if (arrayStart == string::npos ||
        arrayEnd == string::npos)
    {
        throw runtime_error("Invalid roads array");
    }

    string roadsData =
        content.substr(arrayStart + 1,
                       arrayEnd - arrayStart - 1);

    // Read each road object
    regex roadPattern(R"(\{([^{}]*)\})");

    auto begin = sregex_iterator(
        roadsData.begin(), roadsData.end(), roadPattern
    );

    auto end = sregex_iterator();

    regex srcPattern(R"("src"\s*:\s*(\d+))");
    regex destPattern(R"("dest"\s*:\s*(\d+))");
    regex timePattern(R"("time"\s*:\s*([0-9]+(?:\.[0-9]+)?))");
    regex hazardPattern(R"("hazard"\s*:\s*([0-9]+(?:\.[0-9]+)?))");
    regex blockedPattern(R"("blocked"\s*:\s*(true|false))");

    for (auto it = begin; it != end; ++it)
    {
        string roadData = (*it)[1].str();

        smatch srcMatch, destMatch, timeMatch;
        smatch hazardMatch, blockedMatch;

        if (!regex_search(roadData, srcMatch, srcPattern) ||
            !regex_search(roadData, destMatch, destPattern) ||
            !regex_search(roadData, timeMatch, timePattern))
        {
            throw runtime_error("Invalid road data");
        }

        int src = stoi(srcMatch[1].str());
        int dest = stoi(destMatch[1].str());

        double time = stod(timeMatch[1].str());
        double hazard = 1.0;
        bool blocked = false;

        if (regex_search(roadData, hazardMatch, hazardPattern))
            hazard = stod(hazardMatch[1].str());

        if (regex_search(roadData, blockedMatch, blockedPattern))
            blocked = (blockedMatch[1].str() == "true");

        if (src < 0 || src >= totalLocations ||
            dest < 0 || dest >= totalLocations ||
            time < 0 || hazard < 0)
        {
            throw runtime_error("Invalid road values");
        }

        Road road(src, dest, time, hazard);
        road.setBlocked(blocked);

        graph.addRoad(road);
    }

    return graph;
}

bool DatabaseManager::saveLocation(
    int id,
    const std::string& name,
    double latitude,
    double longitude
) {
    try {
        auto collection = database["locations"];

        auto filter = make_document(kvp("id", id));
        auto replacement = make_document(
            kvp("id", id),
            kvp("name", name),
            kvp("latitude", latitude),
            kvp("longitude", longitude)
        );

        mongocxx::options::replace options;
        options.upsert(true);

        collection.replace_one(filter.view(), replacement.view(), options);
        return true;
    } catch (const mongocxx::exception& e) {
        std::cerr << "Saving location failed: " << e.what() << '\n';
        return false;
    }
}

    int roadId,
    int fromLocation,
    int toLocation,
    double distance,
    int status,
    double cost
) {
    try {
        auto collection = database["roads"];

        auto filter = make_document(kvp("roadId", roadId));
        auto replacement = make_document(
            kvp("roadId", roadId),
            kvp("fromLocation", fromLocation),
            kvp("toLocation", toLocation),
            kvp("distance", distance),
            kvp("status", status),
            kvp("cost", cost),
            kvp("blocked", status == 3)
        );

        mongocxx::options::replace options;
        options.upsert(true);

        collection.replace_one(filter.view(), replacement.view(), options);

        std::cerr << "Saving road failed: " << e.what() << '\n';
        return false;
    }
}

bool DatabaseManager::saveEvacuationLog(
    const std::string& eventType,
    const std::string& details
) {
    try {
        auto collection = database["evacuation_logs"];

        const auto now = std::chrono::system_clock::now();
        const auto timestamp = std::chrono::duration_cast<
            std::chrono::milliseconds
        >(now.time_since_epoch()).count();

        auto document = make_document(
            kvp("eventType", eventType),
            kvp("details", details),
            kvp(
                "timestamp",
                bsoncxx::types::b_int64{
                    static_cast<std::int64_t>(timestamp)
                }
            )
        );

        collection.insert_one(document.view());
        return true;
    } catch (const mongocxx::exception& e) {
        std::cerr << "Saving evacuation log failed: " << e.what() << '\n';
        return false;
    }
}

#include "DisasterManager.h"

#include <iostream>
#include <algorithm>

using namespace std;

void DisasterManager::updateRoad(
    Graph &g, int u, int v, int status, DatabaseManager &db)
{
    if (status < 1 || status > 4) {
        cout << "Invalid road status." << endl;
        return;
    }

    if (!g.updateRoadStatus(u, v, status)) {
        cout << "Road not found or could not be updated." << endl;
        return;
    }

    double time = 0.0;
    double cost = 0.0;
    bool blocked = false;
    int actualStatus = status;

    if (!g.getRoadDetails(u, v, time, blocked, cost, actualStatus)) {
        cout << "Road updated in graph, but details could not be read."
             << endl;
        return;
    }

    int roadId = min(u, v) * 100000 + max(u, v);

    if (db.saveRoad(roadId, u, v, time, actualStatus, cost)) {
        cout << "Road updated and saved to MongoDB." << endl;
    } else {
        cout << "Road updated in graph, but MongoDB save failed." << endl;

bool Graph::updateRoadStatus(int u, int v, int status)
{
    if (u < 0 || u >= static_cast<int>(adjList.size()) ||
        v < 0 || v >= static_cast<int>(adjList.size()))
        return false;

    bool found = false;

    auto update = [&](int src, int dest) {
        for (auto &road : adjList[src]) {
            if (road.getDest() == dest) {
                if (status == 1) {
                    road.setBlocked(false);
                    road.setHazardLevel(1.0);
                } else if (status == 2) {
                    road.setBlocked(false);
                    road.setHazardLevel(2.0);
                } else if (status == 3) {
                    road.setBlocked(true);
                } else if (status == 4) {
                    road.setBlocked(false);
                    road.setHazardLevel(3.0);
                } else {
                    return;
                }
                found = true;
            }
        }
    };

    if (status < 1 || status > 4)
        return false;

    update(u, v);
    update(v, u);

    return found;
}

bool Graph::getRoadDetails(
    int u, int v, double& time, bool& blocked, double& cost, int& status)
{
    if (u < 0 || u >= static_cast<int>(adjList.size()))
        return false;

    for (auto &road : adjList[u]) {
        if (road.getDest() == v) {
            time = road.getTime();
            blocked = road.getBlockedStatus();
            cost = road.getCost();

            if (blocked)
                status = 3;
            else if (time != 0.0 && cost >= time * 2.5)
                status = 4;
            else if (time != 0.0 && cost >= time * 1.5)
                status = 2;
            else
                status = 1;

            return true;
        }
    }
    if (v >= 0 && v < static_cast<int>(adjList.size())) {
    for (auto &road : adjList[v]) {
        if (road.getDest() == u) {
            time = road.getTime();
            blocked = road.getBlockedStatus();
            cost = road.getCost();

            if (blocked)
                status = 3;
            else if (time != 0.0 && cost >= time * 2.5)
                status = 4;
            else if (time != 0.0 && cost >= time * 1.5)
                status = 2;
            else
                status = 1;

            return true;
        }
    }
}
    return false;
}

#include "RouteOptimizer.h"
#include "priorityqueue.h"
#include <algorithm>
#include <cmath>
#include <limits>
#include <vector>

using namespace std;

vector<int> RouteOptimizer::findRoute(Graph &g, int src, int des, double &totalCost)
{
    vector<vector<Road>>& adj = g.getAdjList();
    const int n = static_cast<int>(adj.size());

    totalCost = -1;
    if (src < 0 || des < 0 || src >= n || des >= n)
        return {};

    vector<double> dist(n, numeric_limits<double>::infinity());
    vector<int> parent(n, -1);
    PriorityQueue pq;

    dist[src] = 0.0;
    pq.push(src, 0.0);

    while (!pq.empty()) {
        PQNode cur = pq.pop();
        int u = cur.node;

        if (u < 0 || u >= n)
            continue;
        if (cur.cost != dist[u])
            continue;
        if (u == des)
            break;

        for (auto &road : adj[u]) {
            if (road.getBlockedStatus())
                continue;

            int v = road.getDest();
            if (v < 0 || v >= n)
                continue;

            double edgeCost = road.getCost();
            if (edgeCost < 0)
                continue;

            if (dist[u] + edgeCost < dist[v]) {
                dist[v] = dist[u] + edgeCost;
                parent[v] = u;
                pq.push(v, dist[v]);
            }
        }
    }

    if (!std::isfinite(dist[des]))
        return {};

    vector<int> path;
    for (int cur = des; cur != -1; cur = parent[cur]) {
        path.push_back(cur);
        if (cur == src)
            break;
    }

    if (path.empty() || path.back() != src)
        return {};

    reverse(path.begin(), path.end());
    totalCost = dist[des];
    return path;
}

static size_t writeData(void* ptr, size_t size, size_t nmemb, void* data)
{
    string* response = static_cast<string*>(data);
    response->append(static_cast<char*>(ptr), size * nmemb);
    return size * nmemb;
}

APIClient::APIClient()
    : temp(0), rain(0), wind(0), weatherCode(0)
{
}

bool APIClient::getData(double lat, double lon)
{
    CURL* curl = curl_easy_init();

    if (!curl) {
        cerr << "Curl initialization failed.\n";
        return false;
    }

    string response;

    string url =
        "https://api.open-meteo.com/v1/forecast?latitude=" +
        to_string(lat) +
        "&longitude=" + to_string(lon) +
        "&current=temperature_2m,precipitation,wind_speed_10m,weather_code";

    curl_easy_setopt(curl, CURLOPT_URL, url.c_str());
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, writeData);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response);
    curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, 1L);
    curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 1L);
    curl_easy_setopt(curl, CURLOPT_CONNECTTIMEOUT, 10L);
    curl_easy_setopt(curl, CURLOPT_TIMEOUT, 20L);

    CURLcode result = curl_easy_perform(curl);

    long httpCode = 0;
    curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &httpCode);
    curl_easy_cleanup(curl);

    if (result != CURLE_OK || httpCode != 200) {
        cerr << "Weather API request failed: "
             << (result != CURLE_OK
                     ? curl_easy_strerror(result)
                     : "HTTP " + to_string(httpCode))
             << '\n';
        return false;
    }

    // Extract fields specifically from the "current" JSON object.
    size_t currentPos = response.find("\"current\"");
    if (currentPos == string::npos) {
        cerr << "API response has no current weather data.\n";
        return false;
    }

    size_t objectStart = response.find('{', currentPos);
    size_t objectEnd = response.find('}', objectStart);

    if (objectStart == string::npos || objectEnd == string::npos) {
        cerr << "Invalid current weather JSON object.\n";
        return false;
    }

    string current = response.substr(
        objectStart, objectEnd - objectStart + 1);

    auto readNumber = [&](const string& key, double& value) -> bool {
        size_t pos = current.find("\"" + key + "\"");
        if (pos == string::npos) return false;

        pos = current.find(':', pos);
        if (pos == string::npos) return false;

        const char* start = current.c_str() + pos + 1;
        while (*start == ' ' || *start == '\n' || *start == '\t')
            ++start;

        if (*start == '\0' || *start == 'n') return false;

        char* end = nullptr;
        double parsed = strtod(start, &end);

        if (end == start) return false;

        value = parsed;
        return true;
    };

    double newTemp, newRain, newWind, newCode;

    if (!readNumber("temperature_2m", newTemp) ||
        !readNumber("precipitation", newRain) ||
        !readNumber("wind_speed_10m", newWind) ||
        !readNumber("weather_code", newCode)) {
        cerr << "Could not parse all weather fields from API response.\n";
        return false;
    }

    temp = newTemp;
    rain = newRain;
    wind = newWind;
    weatherCode = static_cast<int>(newCode);

    return true;
}

int APIClient::getStatus()
{
    if (rain >= 10 || wind >= 60)
        return 3;
    if (rain >= 5 || wind >= 40)
        return 4;
    if (rain >= 2 || wind >= 25)
        return 2;

    return 1;
}
